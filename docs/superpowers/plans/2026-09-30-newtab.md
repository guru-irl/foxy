# Liquid Fox New Tab Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A private Firefox add-on that replaces the new tab and home page with a Danfo clock, a Bebas Neue date, the current window's pinned tabs and its tab groups; clicking a tab switches to it and closes the page.

**Architecture:** Plain HTML/CSS/ES modules, no framework, no build step, no background script. Pure logic lives in `src/model.js` and `src/actions.js` (unit-tested with `node --test`); `src/render.js` builds DOM; `src/main.js` wires the `browser` API. Python scripts (standard library only) take headless screenshots, zip the add-on and sign it through the AMO v5 API.

**Tech Stack:** WebExtensions MV3 (Firefox 157), vanilla JS modules, CSS (`light-dark()`, container queries, `oklch()`), Node 26 built-in test runner, Python 3 standard library, headless Firefox for screenshots.

**Spec:** `docs/superpowers/specs/2026-09-30-newtab-design.md`

`src/model.js`:

```js
// Pure functions shared by the page and the Node tests: no browser APIs, no DOM.

const COLORS = ["blue", "cyan", "gray", "green", "orange", "pink", "purple", "red", "yellow"];
const WEEKDAYS = ["SUNDAY", "MONDAY", "TUESDAY", "WEDNESDAY", "THURSDAY", "FRIDAY", "SATURDAY"];
const MONTHS = [
  "JANUARY", "FEBRUARY", "MARCH", "APRIL", "MAY", "JUNE",
  "JULY", "AUGUST", "SEPTEMBER", "OCTOBER", "NOVEMBER", "DECEMBER",
];

/** @typedef {{id: number, title: string, icon: string | null}} TabItem */
/** @typedef {{id: number, title: string, color: string, tabs: TabItem[]}} GroupItem */

/**
 * Sort one window's tabs into pinned tabs and tab groups, in sidebar order.
 * Leaves out ungrouped tabs, hidden tabs, the page's own tab, and groups with
 * nothing left to show.
 * @param {Array<object>} tabs  tabs.Tab objects from one window
 * @param {Array<object>} groups  tabGroups.TabGroup objects from the same window
 * @param {number} selfTabId  the new tab page's own tab id
 * @returns {{pinned: TabItem[], groups: GroupItem[]}}
 */
export function buildModel(tabs, groups, selfTabId) {
  const shown = tabs
    .filter((tab) => tab.id !== selfTabId && !tab.hidden)
    .sort((a, b) => a.index - b.index);

  const members = new Map();
  for (const tab of shown) {
    if (tab.pinned || tab.groupId === undefined || tab.groupId === -1) continue;
    if (!members.has(tab.groupId)) members.set(tab.groupId, []);
    members.get(tab.groupId).push(tab);
  }

  const ordered = groups
    .filter((group) => members.has(group.id))
    .map((group) => ({ group, list: members.get(group.id) }))
    .sort((a, b) => a.list[0].index - b.list[0].index);

  return {
    pinned: shown.filter((tab) => tab.pinned).map(toItem),
    groups: ordered.map(({ group, list }) => ({
      id: group.id,
      title: group.title?.trim() || "Unnamed group",
      color: colorKey(group.color),
      tabs: list.map(toItem),
    })),
  };
}

function toItem(tab) {
  return { id: tab.id, title: tab.title || tab.url || "", icon: tab.favIconUrl || null };
}

/** Map a tabGroups color to the sidebar's dot palette key. */
export function colorKey(color) {
  if (color === "grey") return "gray";
  return COLORS.includes(color) ? color : "gray";
}

/** 24-hour local time, e.g. "09:05". */
export function formatTime(date) {
  return `${pad(date.getHours())}:${pad(date.getMinutes())}`;
}

/** English capitals without the year, e.g. "WEDNESDAY 30 SEPTEMBER". */
export function formatDate(date) {
  return `${WEEKDAYS[date.getDay()]} ${date.getDate()} ${MONTHS[date.getMonth()]}`;
}

/** Milliseconds until the next minute starts (1 to 60000). */
export function msUntilNextMinute(date) {
  return 60000 - (date.getSeconds() * 1000 + date.getMilliseconds());
}

function pad(n) {
  return String(n).padStart(2, "0");
}
```

`src/actions.js`:

```js
/**
 * Switch to a tab, then close the new tab page it was picked from.
 * If the switch fails (the tab closed in the meantime) the page stays open.
 * @param {{tabs: {update: Function, remove: Function}}} api  the WebExtension `browser` object
 * @param {number} targetTabId
 * @param {number} selfTabId  the new tab page's own tab
 * @returns {Promise<boolean>} whether the switch happened
 */
export async function activateTab(api, targetTabId, selfTabId) {
  try {
    await api.tabs.update(targetTabId, { active: true });
  } catch {
    return false;
  }
  try {
    await api.tabs.remove(selfTabId);
  } catch {
    // Already closed (for example by the user). The switch still happened.
  }
  return true;
}
```

```js
// Builds the pinned strip and the group cards. Text goes in with textContent
// only, so a tab title can never be parsed as markup.

/**
 * @param {HTMLElement} root  the #pinned strip
 * @param {Array<{id: number, title: string, icon: string | null}>} pinned
 * @param {(tabId: number) => void} onPick
 */
export function renderPinned(root, pinned, onPick) {
  root.hidden = pinned.length === 0;
  root.replaceChildren(
    ...pinned.map((tab) => {
      const button = tabButton("pin", tab, onPick);
      button.append(icon(tab.icon));
      return button;
    }),
  );
}

/**
 * Redraws all cards, keeping each card's scroll position (by group id).
 * @param {HTMLElement} root  the #groups grid
 * @param {Array<{id: number, title: string, color: string, tabs: Array<{id: number, title: string, icon: string | null}>}>} groups
 * @param {(tabId: number) => void} onPick
 */
export function renderGroups(root, groups, onPick) {
  const scroll = new Map();
  for (const list of root.querySelectorAll(".card-tabs")) {
    scroll.set(list.dataset.groupId, list.scrollTop);
  }
  root.hidden = groups.length === 0;
  root.replaceChildren(...groups.map((group) => card(group, onPick)));
  for (const list of root.querySelectorAll(".card-tabs")) {
    list.scrollTop = scroll.get(list.dataset.groupId) ?? 0;
  }
}

function card(group, onPick) {
  const section = el("section", "card");
  section.dataset.color = group.color;

  const heading = el("h2", "card-title");
  const name = el("span", "card-name");
  name.textContent = group.title;
  heading.append(el("span", "dot"), name);

  const list = el("div", "card-tabs");
  list.dataset.groupId = String(group.id);
  for (const tab of group.tabs) {
    const row = tabButton("row", tab, onPick);
    const label = el("span", "row-label");
    label.textContent = tab.title;
    row.append(icon(tab.icon), label);
    list.append(row);
  }

  section.append(heading, list);
  return section;
}

function tabButton(className, tab, onPick) {
  const button = el("button", className);
  button.type = "button";
  button.title = tab.title;
  button.addEventListener("click", () => onPick(tab.id));
  return button;
}

// A favicon, or the neutral globe when there is none or it fails to load
// (for example chrome:// icons, which an add-on page can't load).
function icon(url) {
  if (!url) return el("span", "icon icon-none");
  const img = el("img", "icon");
  img.alt = "";
  img.draggable = false;
  img.addEventListener("error", () => img.replaceWith(el("span", "icon icon-none")), { once: true });
  img.src = url;
  return img;
}

function el(tag, className) {
  const node = document.createElement(tag);
  node.className = className;
  return node;
}
```

```js
import { activateTab } from "./actions.js";
import { buildModel, formatDate, formatTime, msUntilNextMinute } from "./model.js";
import { renderGroups, renderPinned } from "./render.js";

const api = globalThis.browser;

startClock(document.getElementById("time"), document.getElementById("date"));
startTabs(document.getElementById("pinned"), document.getElementById("groups")).catch((error) => {
  console.error("Liquid Fox New Tab: could not show tabs", error);
});

// Updates on each minute boundary, and again whenever the page becomes visible,
// because Firefox throttles timers in background tabs.
function startClock(timeEl, dateEl) {
  let timer = 0;
  const tick = () => {
    const now = new Date();
    timeEl.textContent = formatTime(now);
    dateEl.textContent = formatDate(now);
    clearTimeout(timer);
    timer = setTimeout(tick, msUntilNextMinute(now));
  };
  document.addEventListener("visibilitychange", () => {
    if (!document.hidden) tick();
  });
  tick();
}

// Shows this window's pinned tabs and tab groups and keeps them current: any
// tab or group change schedules one re-read and redraw per animation frame.
async function startTabs(pinnedEl, groupsEl) {
  const self = await api.tabs.getCurrent();
  let queued = false;

  const pick = async (tabId) => {
    if (!(await activateTab(api, tabId, self.id))) schedule();
  };

  async function refresh() {
    const tabs = await api.tabs.query({ windowId: self.windowId });
    let groups = [];
    try {
      groups = await api.tabGroups.query({ windowId: self.windowId });
    } catch (error) {
      console.error("Liquid Fox New Tab: could not read tab groups", error);
    }
    const model = buildModel(tabs, groups, self.id);
    renderPinned(pinnedEl, model.pinned, pick);
    renderGroups(groupsEl, model.groups, pick);
  }

  function schedule() {
    if (queued) return;
    queued = true;
    requestAnimationFrame(() => {
      queued = false;
      refresh().catch((error) => console.error("Liquid Fox New Tab: refresh failed", error));
    });
  }

  const events = [
    api.tabs.onCreated,
    api.tabs.onRemoved,
    api.tabs.onUpdated,
    api.tabs.onMoved,
    api.tabs.onAttached,
    api.tabs.onDetached,
    api.tabGroups?.onCreated,
    api.tabGroups?.onUpdated,
    api.tabGroups?.onMoved,
    api.tabGroups?.onRemoved,
  ];
  for (const event of events) event?.addListener(schedule);

  await refresh();
}
```

```python
#!/usr/bin/env python3
"""Zip the add-on files into dist/liquidfox-newtab-<version>.zip.

The zip is unsigned; scripts/sign.py turns it into a signed .xpi. File dates
are fixed so the same sources always give the same bytes.
"""
import json
import pathlib
import zipfile

ROOT = pathlib.Path(__file__).resolve().parent.parent
INCLUDE = ["manifest.json", "newtab.html", "newtab.css", "src/*.js", "fonts/*", "icons/*"]
FIXED_TIME = (1980, 1, 1, 0, 0, 0)


def package_files(root=ROOT):
    """Sorted POSIX paths (relative to root) of the files that go into the add-on."""
    found = set()
    for pattern in INCLUDE:
        for path in root.glob(pattern):
            if path.is_file() and not path.name.startswith("."):
                found.add(path.relative_to(root).as_posix())
    return sorted(found)


def build(root=ROOT, out_dir=None):
    version = json.loads((root / "manifest.json").read_text())["version"]
    out_dir = out_dir or root / "dist"
    out_dir.mkdir(parents=True, exist_ok=True)
    out = out_dir / f"liquidfox-newtab-{version}.zip"
    with zipfile.ZipFile(out, "w", zipfile.ZIP_DEFLATED) as zf:
        for name in package_files(root):
            info = zipfile.ZipInfo(name, FIXED_TIME)
            info.compress_type = zipfile.ZIP_DEFLATED
            info.external_attr = 0o644 << 16
            zf.writestr(info, (root / name).read_bytes())
    return out


if __name__ == "__main__":
    print(build())
```

```python
#!/usr/bin/env python3
"""Sign the add-on as an unlisted addons.mozilla.org add-on and download the .xpi.

Needs ~/.config/liquidfox-newtab/amo.env, readable only by you (chmod 600):

    AMO_JWT_ISSUER=user:12345:67
    AMO_JWT_SECRET=...

Create the key and secret at https://addons.mozilla.org/developers/addon/api/key/.

Usage: python3 scripts/build.py && python3 scripts/sign.py
Output: dist/liquidfox-newtab-<version>-signed.xpi
"""
import base64
import hashlib
import hmac
import json
import pathlib
import stat
import sys
import time
import urllib.error
import urllib.parse
import urllib.request
import uuid

ROOT = pathlib.Path(__file__).resolve().parent.parent
API = "https://addons.mozilla.org/api/v5"
CREDENTIALS = pathlib.Path.home() / ".config" / "liquidfox-newtab" / "amo.env"
USER_AGENT = "liquidfox-newtab-sign/1.0"


def b64url(data):
    return base64.urlsafe_b64encode(data).rstrip(b"=").decode("ascii")


def make_jwt(issuer, secret, now=None, jti=None):
    """An HS256 JWT with the claims AMO wants: iss, jti, iat and exp (60 s later)."""
    now = int(time.time()) if now is None else now
    header = {"alg": "HS256", "typ": "JWT"}
    payload = {"iss": issuer, "jti": jti or str(uuid.uuid4()), "iat": now, "exp": now + 60}
    signing_input = ".".join(
        b64url(json.dumps(part, separators=(",", ":")).encode()) for part in (header, payload)
    )
    signature = hmac.new(secret.encode(), signing_input.encode("ascii"), hashlib.sha256).digest()
    return f"{signing_input}.{b64url(signature)}"


def read_credentials(path=CREDENTIALS):
    """AMO_JWT_ISSUER and AMO_JWT_SECRET from a KEY=VALUE file only its owner can read."""
    if not path.exists():
        raise SystemExit(
            f"Missing {path}. Create it with AMO_JWT_ISSUER=... and AMO_JWT_SECRET=... (chmod 600)."
        )
    if path.stat().st_mode & (stat.S_IRWXG | stat.S_IRWXO):
        raise SystemExit(f"{path} is readable by other users. Run: chmod 600 '{path}'")
    values = {}
    for line in path.read_text().splitlines():
        line = line.strip()
        if not line or line.startswith("#") or "=" not in line:
            continue
        key, value = line.split("=", 1)
        values[key.strip()] = value.strip().strip("\"'")
    missing = [key for key in ("AMO_JWT_ISSUER", "AMO_JWT_SECRET") if not values.get(key)]
    if missing:
        raise SystemExit(f"{path} is missing {', '.join(missing)}")
    return values["AMO_JWT_ISSUER"], values["AMO_JWT_SECRET"]


def encode_multipart(fields, files, boundary=None):
    """fields {name: str}, files {name: (filename, bytes, content_type)} -> (body, content_type)."""
    boundary = boundary or uuid.uuid4().hex
    lines = []
    for name, value in fields.items():
        lines += [
            f"--{boundary}".encode(),
            f'Content-Disposition: form-data; name="{name}"'.encode(),
            b"",
            value.encode(),
        ]
    for name, (filename, data, content_type) in files.items():
        lines += [
            f"--{boundary}".encode(),
            f'Content-Disposition: form-data; name="{name}"; filename="{filename}"'.encode(),
            f"Content-Type: {content_type}".encode(),
            b"",
            data,
        ]
    lines += [f"--{boundary}--".encode(), b""]
    return b"\r\n".join(lines), f"multipart/form-data; boundary={boundary}"


class Amo:
    """Minimal AMO v5 client; every request carries a fresh JWT."""

    def __init__(self, issuer, secret):
        self.issuer = issuer
        self.secret = secret

    def request(self, method, url, body=None, content_type=None):
        headers = {
            "Authorization": f"JWT {make_jwt(self.issuer, self.secret)}",
            "User-Agent": USER_AGENT,
            "Accept": "application/json",
        }
        if content_type:
            headers["Content-Type"] = content_type
        req = urllib.request.Request(url, data=body, headers=headers, method=method)
        try:
            with urllib.request.urlopen(req, timeout=120) as resp:
                return resp.read()
        except urllib.error.HTTPError as err:
            detail = err.read().decode("utf-8", "replace")[:2000]
            raise SystemExit(f"{method} {url} failed: HTTP {err.code}\n{detail}")

    def json(self, method, url, payload=None, body=None, content_type=None):
        if payload is not None:
            body, content_type = json.dumps(payload).encode(), "application/json"
        return json.loads(self.request(method, url, body, content_type) or b"{}")


def wait(check, what, interval, timeout=900):
    """Call check() until it returns something other than None."""
    deadline = time.monotonic() + timeout
    while True:
        result = check()
        if result is not None:
            return result
        if time.monotonic() > deadline:
            raise SystemExit(f"Timed out waiting for {what}.")
        print(f"  waiting for {what}...")
        time.sleep(interval)


def sign(amo, xpi, guid, version, out, interval=5):
    """Upload (unlisted), wait for validation, submit, wait for signing, download."""
    guid_path = urllib.parse.quote(guid, safe="@")

    print(f"Uploading {xpi.name} (unlisted)...")
    body, content_type = encode_multipart(
        {"channel": "unlisted"},
        {"upload": (xpi.name, xpi.read_bytes(), "application/zip")},
    )
    upload = amo.json("POST", f"{API}/addons/upload/", body=body, content_type=content_type)
    upload_url = f"{API}/addons/upload/{upload['uuid']}/"

    def validated():
        current = amo.json("GET", upload_url)
        return current if current.get("processed") else None

    upload = wait(validated, "validation", interval)
    for message in (upload.get("validation") or {}).get("messages", []):
        print(f"  {message.get('type', '?')}: {message.get('message', '')}")
    if not upload.get("valid"):
        raise SystemExit("AMO validation failed; see the messages above.")

    print("Submitting the version...")
    amo.json("PUT", f"{API}/addons/addon/{guid_path}/", payload={"version": {"upload": upload["uuid"]}})
    version_url = f"{API}/addons/addon/{guid_path}/versions/v{version}/"

    def signed():
        file = amo.json("GET", version_url).get("file") or {}
        if file.get("status") == "disabled":
            raise SystemExit("AMO rejected or disabled this version.")
        return file if file.get("status") == "public" and file.get("url") else None

    file = wait(signed, "signing", interval)
    out.write_bytes(amo.request("GET", file["url"]))
    print(f"Signed: {out}")


def main():
    manifest = json.loads((ROOT / "manifest.json").read_text())
    version = manifest["version"]
    guid = manifest["browser_specific_settings"]["gecko"]["id"]
    xpi = ROOT / "dist" / f"liquidfox-newtab-{version}.zip"
    if not xpi.exists():
        raise SystemExit(f"Missing {xpi}. Run: python3 scripts/build.py")
    issuer, secret = read_credentials()
    sign(Amo(issuer, secret), xpi, guid, version, ROOT / "dist" / f"liquidfox-newtab-{version}-signed.xpi")


if __name__ == "__main__":
    main()
```
